# 当前因为仅仅支持浮点类型，所以下面的很多运算定义可能存在异常未处理
# 但是此时只关注编译器的正确性，运算定义正确性暂时忽略

#自定义二元逻辑或（非内置）
def binary | 2 (lhs, rhs) if lhs then 1 else if rhs then 1 else 0;

#自定义二元取模（非内置）
def binary % 6 (lhs, rhs) lhs - (lhs / rhs) * rhs;

#自定义一元的取反，虽然和内置的`-`符号一样，但是因为一元运算符不记录到BinOpMap中，不会冲突.
def  unary-(v) 0 - v;

#自定义一元按位取反（近似）
def unary ~ (v) -v - 1;

#自定义一元转换为布尔值（非零→1，零→0）
def unary $ (v) if v then 1 else 0;

# 测试表达式（无变量，纯嵌套）
extern printnum(n);

#printnum((5 | 3) + (10 % 4) + (~ 8) + ($ (~ 8)) + (if (1 | 0) then 10 else 20));

# 直接计算并输出结果
printnum(
    (5 | 3)                     # 逻辑或 => 1
    + (10 % 4)                  # 取模 => 2
    + (~ 8)                     # 按位取反 => -9
    + ($ (~ 8))                 # 转换为布尔 => 1（非零）
    + (if (1 | 0) then 10 else 20)  # if 表达式 => 10
);

# ========== var 变量声明测试 ==========

# 1. 最基础：单变量声明
def testVar1() var x = 10 in x + 1;

# 2. 多变量同时声明
def testVar2() var a = 1, b = 2 in a + b;

# 3. 变量依赖：后面变量引用前面变量
def testVar3() var a = 3, b = a * 2 in a + b;

# 4. 变量参与复杂表达式
def testVar4(x) var y = x * 2, z = y + 1 in z / 3;

# 5. 变量与 if 结合
def testVar5(x) var t = x < 0 in if t then 0-x else x;

# 6. 变量与 for 循环结合
def testVar6() var s = 0 in for i = 1, i < 5 in s = s + i;

# 7. 嵌套 var（作用域遮蔽）
def testVar7()
    var x = 10 in
        var x = 100 in
            x + 1;

# 8. 变量与自定义运算符结合
def testVar8()
    var a = 5 | 0, b = 10 % 3 in
        a + b;

# 9. 变量存储函数调用结果
def testVar9(x) var s = sin(x), c = cos(x) in s * s + c * c;

# 10. 多语句函数体中使用 var（如果语法支持分号分隔表达式序列）
def testVar10(x)
    var a = x + 1 in
        var b = a * 2 in
            b - 1;

# 11. 变量默认值与参数互动
def testVar11(a, b)
    var min = if a < b then a else b,
        max = if a < b then b else a in
        max - min;

# 12. 用于累加/计数场景
def sumTo(n)
    var total = 0 in
        for i = 1, i < n+1 in total = total + i;

# 13. 变量与一元运算符
def testVar13()
    var x = 8 in
        (~ x) + ($ x);

# 14. 浮点变量声明
def testVar14()
    var pi = 3.14159, r = 2.0 in
        pi * r * r;
