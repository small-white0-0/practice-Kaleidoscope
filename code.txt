# 当前因为仅仅支持浮点类型，所以下面的很多运算定义可能存在异常未处理
# 但是此时只关注编译器的正确性，运算定义正确性暂时忽略

#自定义二元逻辑或（非内置）
def binary | 2 (lhs, rhs) if lhs then 1 else if rhs then 1 else 0;

#自定义二元取模（非内置）
def binary % 6 (lhs, rhs) lhs - (lhs / rhs) * rhs;

#自定义一元的取反，虽然和内置的`-`符号一样，但是因为一元运算符不记录到BinOpMap中，不会冲突.
def  unary-(v) 0 - v;

#自定义一元按位取反（近似）
def unary ~ (v) -v - 1;

#自定义一元转换为布尔值（非零→1，零→0）
def unary $ (v) if v then 1 else 0;

# 测试表达式（无变量，纯嵌套）
extern printnum(n);

#printnum((5 | 3) + (10 % 4) + (~ 8) + ($ (~ 8)) + (if (1 | 0) then 10 else 20));

# 直接计算并输出结果
printnum(
    (5 | 3)                     # 逻辑或 => 1
    + (10 % 4)                  # 取模 => 2
    + (~ 8)                     # 按位取反 => -9
    + ($ (~ 8))                 # 转换为布尔 => 1（非零）
    + (if (1 | 0) then 10 else 20)  # if 表达式 => 10
);
# 整个表达式计算结果：1 + (-11) + 2 + (-9) + 1 + 10 = -6
